#===== Identification of Y-linked scaffolds

git clone https://github.com/isabela42/OligoY.git # replace ipda_target-inference_YGS in OligoY/scripts/ with version (v2)
chmod +x OligoY/scripts/CarvalhoClark2013/*  # give permissions
conda create -n jellyfishV226_env -c bioconda -c conda-forge jellyfish=2.2.6
perl -MCPAN -e 'install PerlIO::gzip' # install perl modules
perl -MCPAN -e 'install Bit::Vector' 
perl -MCPAN -e 'install Carp::Clan'

# For D. miranda
#Download male and female reads
fastq-dump --gzip SRR364796 &
fastq-dump --gzip SRR789669 &

# Run Y identification
# Using 15mer
nohup bash OligoY/scripts/ipda_target-inference_YGSv2.sh -k 18 -y SRR364796.fastq.gz -x SRR789669.fastq.gz -t Dmir106.fasta -q 10 -m 2 -f 4 -g 2 -v 20 -p 75 -o YGS_result15/ 

# Using 18mer
nohup bash OligoY/scripts/ipda_target-inference_YGSv2.sh -k 15 -y SRR364796.fastq.gz -x SRR789669.fastq.gz -t Dmir106.fasta -q 10 -m 2 -f 4 -g 2 -v 20 -p 75 -o YGS_result18/
